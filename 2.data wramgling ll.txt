import pandas as pd
import numpy as np

data = pd.read_csv("academic.csv")
print(data.isnull().sum())

# Fill missing values
data.fillna(data.mean(numeric_only=True), inplace=True)

# Detect outliers using IQR
Q1 = data['Marks'].quantile(0.25)
Q3 = data['Marks'].quantile(0.75)
IQR = Q3 - Q1
data = data[(data['Marks'] >= Q1 - 1.5 * IQR) & (data['Marks'] <= Q3 + 1.5 * IQR)]

# Log transform for normality
data['Marks'] = np.log1p(data['Marks'])



Q1: What is an outlier?
Ans: An outlier is a data point that is far from other values. It may affect the result and needs to be removed or treated.

Q2: What is data transformation?
Ans: It is the process of applying mathematical functions (like log, sqrt) to change data scale, reduce skewness, or improve model performance.
